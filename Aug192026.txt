$headers = @{
    "Content-Type" = "application/json"
    "api-key" = "<your-api-key>"
}

    messages = @(
        @{ role = "system"; content = "You are a helpful assistant." },
        @{ role = "user"; content = "Say hello in one sentence." }
    )
    max_tokens = 50
} | ConvertTo-Json -Depth 3
